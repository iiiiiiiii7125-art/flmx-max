// L2 scratch-only reproduction (NOT part of the audited tree):
// update*/delete* nil-item panic + loader nil-item panic.
// Each test recovers the panic and FAILs with the panic value when one
// occurs (proving the defect), or reports PASS when no panic occurs.
package socket

import (
	"testing"

	"github.com/go-gost/x/config"
	"github.com/go-gost/x/registry"
)

// runAndCatch runs f, recovering any panic; returns (panicked, value).
func runAndCatch(f func()) (panicked bool, value any) {
	defer func() {
		if r := recover(); r != nil {
			panicked = true
			value = r
		}
	}()
	f()
	return false, nil
}

func TestL2UpdateChainNilPanics(t *testing.T) {
	orig := config.Global()
	t.Cleanup(func() { config.Set(orig) })
	registry.ChainRegistry().Unregister("l2-chain")
	registry.ChainRegistry().Unregister("l2-target")
	// nil item BEFORE the target: updateChain's loop derefs c.Chains[i].Name.
	valid := config.ChainConfig{Name: "l2-target", Hops: []*config.HopConfig{{Name: "h", Nodes: []*config.NodeConfig{{Name: "n", Addr: "127.0.0.1:443", Connector: &config.ConnectorConfig{Type: "relay"}, Dialer: &config.DialerConfig{Type: "tcp"}}}}}}
	config.Set(&config.Config{Chains: []*config.ChainConfig{
		nil,
		{Name: "l2-target", Hops: []*config.HopConfig{{Name: "h", Nodes: []*config.NodeConfig{{Name: "n", Addr: "127.0.0.1:443"}}}}},
	}})
	panicked, v := runAndCatch(func() {
		_ = updateChain(updateChainRequest{Chain: "l2-target", Data: valid})
	})
	if !panicked {
		t.Logf("updateChain did NOT panic on nil chain entry")
	} else {
		t.Errorf("updateChain PANICKED on nil chain entry: %v", v)
	}
}
